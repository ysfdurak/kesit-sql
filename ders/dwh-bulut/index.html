Zaman yolculuğu ve klon

-- BigQuery: tablonun 1 saat önceki hali
SELECT * FROM anadolu.orders
FOR SYSTEM_TIME AS OF TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 1 HOUR);

-- Snowflake: yanlışlıkla silinen tabloyu geri getir, sıfır kopya klon
UNDROP TABLE orders;
CREATE TABLE orders_test CLONE orders;

-- Databricks (Delta Lake): sürüm geçmişi ve eski sürüm
DESCRIBE HISTORY orders;
SELECT * FROM orders VERSION AS OF 12;
